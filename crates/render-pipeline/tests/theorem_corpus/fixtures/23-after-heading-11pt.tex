\section{Introduction}
\begin{theorem}
There are infinitely many prime numbers, and this sentence is long enough that it has to wrap onto a second line of the paragraph.
\end{theorem}
\subsection{Proof}
\begin{proof}
Let $n$ be a positive integer and consider the product of all primes up to $n$ plus one; none of the listed primes divides it, which is the classical argument.
\end{proof}
